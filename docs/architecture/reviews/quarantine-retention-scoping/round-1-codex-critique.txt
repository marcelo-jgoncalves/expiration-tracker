 * The single real trigger surface for tenant deletion: moves the tenant's `TenantLifecycleRecord`

 * `purgeTenant()` were real, working, and completely unreachable â€” nothing in production ever
 * called either.
 *
 * Step ordering is load-bearing and was corrected during the design protocol (Rodada 3 Fix 8): the
 * terminal-state check runs BEFORE the unconditional `StartExecution`, not after. Rodada 2's
 * original wording implied any post-`ACTIVE` status should trigger a fresh launch, which would
 * have re-launched executions for VERIFIED/DELETED/BLOCKED/HELD tenants â€” states where a purge
 * execution is either finished or deliberately parked awaiting an operator.

/** Same narrow read contract the orchestrator's transition handler uses â€” `ConsistentRead: true`
 * is mandatory, since this read produces the `expectedVersion` the OCC-fenced transition below is
 * conditioned on. */
export interface TenantLifecycleReader {
  read(tenantId: string): Promise<TenantLifecycleRecord | undefined>;
}

/** States where a purge is either already finished or parked awaiting a human â€” closing is
 * refused outright rather than re-launched. Named "unavailable", not "terminal": an operator can
 * still remediate BLOCKED/HELD (`tenant-lifecycle-record.ts`), so these are not all true terminal
 * states. */
const CLOSURE_UNAVAILABLE_STATUSES: ReadonlySet<TenantLifecycleStatus> = new Set(["VERIFIED", "DELETED", "BLOCKED", "HELD"]);

/** The genuinely in-flight states â€” a close() call landing here performs no write and only
 * repeats the idempotent launch. */
const IN_FLIGHT_STATUSES: ReadonlySet<TenantLifecycleStatus> = new Set(["DELETING", "QUIESCING", "PURGING"]);

export interface CloseOrganizationResult {
  tenantId: string;
  status: TenantLifecycleStatus;
  /** False when the record was already DELETING+ and this call only repeated the launch. */
  transitioned: boolean;
}

export class CloseOrganizationService {
  constructor(
    private readonly store: SystemMutationStore,
    private readonly reader: TenantLifecycleReader,
    private readonly executions: TenantPurgeExecutionStarter,
    private readonly tableName: string,
    private readonly now: () => string = () => new Date().toISOString(),
  ) {}

  async close(ctx: RequestContext): Promise<CloseOrganizationResult> {
    const tenantId = ctx.tenant.tenantId;
    authorize({ context: ctx, action: "organization:close", resource: { tenantId } });

    const record = await this.reader.read(tenantId);
    if (!record) {
      throw new NotFoundError("Organization lifecycle record not found.", { organizationId: tenantId });
    }
    if (CLOSURE_UNAVAILABLE_STATUSES.has(record.status)) {
      throw new OrganizationClosureUnavailableError(undefined, { organizationId: tenantId, status: record.status });
    }

    let transitioned = false;
    if (record.status === "ACTIVE") {
      try {
        await transitionTenantLifecycle({
          store: this.store,
          tableName: this.tableName,
          tenantId,
          from: "ACTIVE",
          to: "DELETING",
          expectedVersion: record.version,
          now: this.now,
        });
        transitioned = true;
      } catch (err) {
        if (!(err instanceof SystemMutationConflictError)) throw err;
        // A concurrent close() (double-click, retried request) won the race. Re-read once: if the
        // record legitimately landed on an in-flight state, this call simply joins that closure
        // and proceeds to the shared idempotent launch below. Anything else â€” still ACTIVE, or
        // raced into a stuck state â€” is genuinely unexpected and is not papered over.
        const after = await this.reader.read(tenantId);
        if (!after || !IN_FLIGHT_STATUSES.has(after.status)) throw err;
      }
    } else if (!IN_FLIGHT_STATUSES.has(record.status)) {
      // Exhaustiveness guard: ACTIVE, the 4 unavailable states, and the 3 in-flight states cover
      // every TenantLifecycleStatus. Reaching here means a status was added without deciding what
      // closing should do with it â€” fail loudly rather than silently launching an execution.
      throw new OrganizationClosureUnavailableError(undefined, { organizationId: tenantId, status: record.status });
    }

    await this.executions.startExecution({ name: tenantId, input: { tenantId } });

    return { tenantId, status: transitioned ? "DELETING" : record.status, transitioned };
  }
}

import type { EntityKey } from "../dynamodb/occ.js";

/**
 * Forward-only state machine (approved design Â§G):
 *   ACTIVE -> DELETING -> QUIESCING -> PURGING -> VERIFIED -> DELETED
 * Any of DELETING/QUIESCING/PURGING/VERIFIED can additionally move to BLOCKED/HELD (stuck
 * state â€” legal hold or a purge error that keeps failing) and back to the SAME state once
 * remediated. ACTIVE is never re-entered from any other state (no "un-delete") and DELETED
 * is a true terminal â€” nothing transitions out of it (the sweeper described in the design
 * doc repairs residue in place, it never re-opens the lifecycle record's status).
 */
export type TenantLifecycleStatus =
  | "ACTIVE"
  | "DELETING"
  | "QUIESCING"
  | "PURGING"
  | "VERIFIED"
  | "DELETED"
  | "BLOCKED"
  | "HELD";

export interface TenantLifecycleRecord extends EntityKey {
  SK: "LIFECYCLE";
  entityType: "TenantLifecycleRecord";
  tenantId: string;
  status: TenantLifecycleStatus;
  /** Set when entering BLOCKED/HELD and cleared on remediation back to the prior state â€” a
   * short machine-readable reason for the stuck state (e.g. "PURGE_S3_ERRORS",
   * "LEGAL_HOLD"), never free text containing tenant PII. */
  blockedReason?: string;

   * required whenever status is BLOCKED/HELD, absent otherwise. */

/**
 * Statuses under which a `TenantBusinessMutation` may be admitted â€” deliberately just
 * `ACTIVE` (approved design Â§H invariant 1: "Nenhuma mutaÃ§Ã£o de negÃ³cio tenant-scoped
 * commita em DynamoDB sem ConditionCheck de TenantLifecycleRecord.status = ACTIVE").
 */
export const TENANT_ACTIVE_STATUS: TenantLifecycleStatus = "ACTIVE";

/** Forward-only transition graph. A state maps to the set of states it may legally move to
 * next; BLOCKED/HELD's own allowed targets depend on `blockedFrom` (checked separately by
 * `canResumeFromBlocked`, since the graph itself can't express "return to wherever you came
 * from" as a fixed edge list). */
const FORWARD_TRANSITIONS: Record<TenantLifecycleStatus, ReadonlySet<TenantLifecycleStatus>> = {
  ACTIVE: new Set(["DELETING"]),
  DELETING: new Set(["QUIESCING", "BLOCKED", "HELD"]),
  QUIESCING: new Set(["PURGING", "BLOCKED", "HELD"]),
  PURGING: new Set(["VERIFIED", "BLOCKED", "HELD"]),
  VERIFIED: new Set(["DELETED", "BLOCKED", "HELD"]),
  DELETED: new Set([]),
  BLOCKED: new Set([]), // only canResumeFromBlocked() below, not a fixed forward edge
  HELD: new Set([]),
};

/**
 * True if `from -> to` is a legal forward transition, OR a remediation resume out of
 * BLOCKED/HELD back to the exact state it was blocked from. Never true for anything that
 * would re-enter ACTIVE or leave DELETED â€” those two guarantees are the entire point of
 * this fence (approved design Â§H invariant 4, Â§Q "no ACTIVE reversion, no DELETED
 * resurrection").
 */
export function canTransition(from: TenantLifecycleStatus, to: TenantLifecycleStatus, blockedFrom?: TenantLifecycleStatus): boolean {
  if (to === "ACTIVE") return false; // never re-enter ACTIVE from anywhere
  if (from === "DELETED") return false; // DELETED is terminal, nothing leaves it
  if ((from === "BLOCKED" || from === "HELD") && blockedFrom) {
    return to === blockedFrom;
  }
  return FORWARD_TRANSITIONS[from]?.has(to) ?? false;
}

export class InvalidTenantLifecycleTransitionError extends Error {
  constructor(
    public readonly from: TenantLifecycleStatus,
    public readonly to: TenantLifecycleStatus,
  ) {
    super(`Invalid TenantLifecycleRecord transition: ${from} -> ${to}`);
    this.name = "InvalidTenantLifecycleTransitionError";
  }
}

/** Throws InvalidTenantLifecycleTransitionError instead of returning false â€” for callers
 * (e.g. the future lifecycle-transition worker) that want a fail-fast assertion rather than
 * a boolean they must remember to check. */
export function assertValidTransition(from: TenantLifecycleStatus, to: TenantLifecycleStatus, blockedFrom?: TenantLifecycleStatus): void {
  if (!canTransition(from, to, blockedFrom)) {
    throw new InvalidTenantLifecycleTransitionError(from, to);
  }
}

78:## Proposta
79:
80:### 1. Mecanismo de quarentena (cross-cutting)
81:
82:**Novo estado `HELD_FOR_RECOVERY`, inserido entre `DELETING` e `QUIESCING`** no lifecycle de
83:qualquer entidade que declare "dado persistido + exclusão irreversível" — hoje só
84:`TenantLifecycleRecord`/W3-07 tem esse lifecycle real; o desenho é feito para ser o padrão que a
85:futura feature de arquivos (e qualquer feature seguinte da mesma classe) reusa sem reinventar.
86:
87:- **Transição**: `ACTIVE → HELD_FOR_RECOVERY` (era `ACTIVE → DELETING` direto). `DELETING` passa a
88:  significar exclusividade "o operador confirmou, quarentena venceu, purga física em curso" — não
89:  mais o primeiro passo.
90:- **Duração**: **30 dias** — dentro da faixa convergente da pesquisa (20-90d), no ponto médio,
91:  consistente com o próprio `privacy-lgpd.md` §4 já usar "30 dias" como prazo padrão default em 2
92:  das 9 classes (`ACCOUNT_ACTIVE`, `CORE_USER_DATA`) — reuso de um número que já é norma interna do
93:  projeto, não um valor novo importado sem conexão com o resto do sistema.
94:- **Ação real de cancelamento**: **existe** — novo `CancelOrganizationClosureService`/ação
95:  `organization:cancel-close` (mesmo tier `OWNER_ROLES` de `organization:close`), transição
96:  `HELD_FOR_RECOVERY → ACTIVE` (única exceção ao "ACTIVE nunca é reentrada" documentado em
97:  `tenant-lifecycle-record.ts` — o comentário original previu certo o desenho pré-quarentena, esta
98:  rodada o substitui deliberadamente, não o contradiz por descuido). Sem essa ação real, a
99:  "quarentena" seria só um atraso, não uma janela de recuperação de fato — o próprio motivo que
100:  levou Marcelo a levantar o achado.
101:- **Implementação AWS**: reusa o mesmo Step Functions já existente (`tenant-purge.asl.json`) — o
102:  `Wait(1800s)` atual desloca de posição (passa a rodar DEPOIS de `HELD_FOR_RECOVERY→DELETING`,
103:  mantendo seu propósito operacional intacto) e um novo `Wait` de 30 dias (`Timestamp`/`Seconds:
104:  2592000` — Step Functions suporta waits de até 1 ano) entra antes. Nenhum novo serviço AWS.
105:- **Generalização**: o par de estados (`HELD_FOR_RECOVERY` + duração configurável + ação de cancel)
106:  é documentado como o padrão a seguir por qualquer lifecycle futuro de exclusão irreversível — não
107:  uma classe TypeScript compartilhada prematura (a feature de arquivos nem existe ainda para saber
108:  a forma real do reuso de código; documentar o PADRÃO agora, herdar a implementação depois, é
109:  proporcional ao que se sabe hoje — mesmo raciocínio já usado em B2B-3 para adiar
110:  `ownerCount`).

112:### 2. Priorização das 7 classes LGPD sem purga física real
113:
114:Ordem de prioridade (maior exposição/sensibilidade primeiro), critério: dado pessoal
115:identificável diretamente exposto a um titular vs. dado agregado/técnico:
116:
117:1. **`CORE_USER_DATA`** (itens/políticas/ocorrências dentro de tenant `ACTIVE`) — maior volume de
118:   dado pessoal do produto, sem purga por idade hoje fora do fechamento de tenant.
119:2. **`DELIVERY_RECORD`** (intents/attempts, 180 dias) — prazo já definido e vencendo continuamente
120:   desde o M10 (import) e M7 (extração) irem a produção; sem worker, o gap cresce todo dia.
121:3. **`SECURITY_AUDIT`** (AuditEvent/logs, 365 dias) — maior volume acumulado ao longo do tempo
122:   (evento por mutação sensível), mas exposição pessoal menor que 1/2 (metadado de auditoria, não
123:   o próprio dado de negócio).
124:4. **`LEGAL_EVIDENCE`** — sensibilidade alta, mas propositalmente a mais lenta de implementar

88: * this fence (approved design §H invariant 4, §Q "no ACTIVE reversion, no DELETED

190:/**
191: * authorize() — implementation-blueprint.md §4.3: "authorize() verifica primeiro
192: * igualdade de tenant, depois papel e vínculo por recurso." Throws
193: * AuthorizationDeniedError (never returns false) so callers cannot forget to check
194: * a boolean result — the http layer maps this to AuthorizationError (403).
195: *
196: * Order matters and is deliberately fail-closed at every step:
197: *  1. context must carry at least one role (no membership => no access at all).
198: *  2. if a resource is supplied, resource.tenantId MUST equal context.tenant.tenantId —
199: *     this is the check that makes cross-tenant ID substitution fail, independent of
200: *     role, and independent of what the caller *claims* the resource's tenant is (the
201: *     resource object here must always be populated from a DB read keyed by the
202: *     authenticated tenantId, never from client input — see repository contract).
203: *  3. role must be sufficient for the action.
204: *  4. if the resource declares an owner/assignee, membership role alone isn't enough
205: *     for actions that are resource-scoped by ownership in the future; MVP has no
206: *     per-item ownership finer than tenant, so this step is a no-op today but kept
207: *     explicit so a future item-level ACL doesn't require re-deriving this function.
208: */
209:export function authorize(input: AuthorizationInput): void {
210:  const { context, action, resource } = input;
211:  const roles = context.tenant.roles as Role[];
212:
213:  if (roles.length === 0) {
214:    throw new AuthorizationDeniedError("NO_MEMBERSHIP", action);
215:  }
216:
217:  if (resource && resource.tenantId !== context.tenant.tenantId) {
218:    throw new AuthorizationDeniedError("TENANT_MISMATCH", action);
219:  }
220:
221:  const allowedRoles = ACTION_ROLES[action];
222:  const hasRole = roles.some((role) => allowedRoles.has(role));
223:  if (!hasRole) {
224:    throw new AuthorizationDeniedError("INSUFFICIENT_ROLE", action);
225:  }
226:
227:  if (resource?.assigneeUserId && resource.ownerUserId) {
228:    const isOwnerOrAssignee =
229:      resource.ownerUserId === context.principal.userId ||
230:      resource.assigneeUserId === context.principal.userId;
231:    // OWNER and ADMIN bypass per-resource ownership (tenant-wide content admin, B2B-7 -
232:    // ADMIN has parity with OWNER over business resources, see authorization.ts role model
233:    // comment above), MEMBER/VIEWER scoped resources require ownership match once per-item
234:    // ACLs exist. This is enforced only when both fields ARE present (defensive for future
235:    // use), never invented from nothing.
236:    if (!roles.includes("OWNER") && !roles.includes("ADMIN") && !isOwnerOrAssignee) {
237:      throw new AuthorizationDeniedError("RESOURCE_OWNERSHIP_MISMATCH", action);
238:    }
239:  }

2. Falta toda a coordenação entre cancelamento e a execução em espera.

A proposta manda iniciar o workflow, esperar 30 dias e permitir `HELD_FOR_RECOVERY → ACTIVE`, mas não diz como a execução Step Functions é interrompida. Se apenas o registro voltar a `ACTIVE`, a execução continua viva. Ao acordar, tentará avançar para `DELETING`.

O OCC provavelmente impede a purga naquele momento, mas isso não torna o fluxo correto: a execução cairá no caminho de erro e tentará marcar o tenant como bloqueado usando um `from` que já não corresponde ao estado real. O resultado provável é falha secundária ao marcar `BLOCKED` e uma execução `Failed`, produzindo alarme falso para um cancelamento bem-sucedido.

Faltam decisões sobre:

- `StopExecution`;
- ordem entre parar a execução e restaurar `ACTIVE`;
- falha de `StopExecution` depois da restauração;
- cancelamento concorrente com o vencimento exato dos 30 dias;
- idempotência de dois cancelamentos;
- prova de que nenhuma purga pode começar depois de uma resposta HTTP de cancelamento bem-sucedida.

Essa é a parte mais importante do design, e está ausente.

3. O nome fixo da execução quebra o ciclo cancelar → fechar novamente.

O código inicia a execução com `name: tenantId` em [close-organization.ts:107](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/modules/organization/application/close-organization.ts:107). Se a execução for interrompida para cancelar, esse nome não fica imediatamente reutilizável. Portanto, um OWNER que cancelar e depois fechar novamente pode não conseguir iniciar uma nova execução.

A proposta precisa introduzir identidade por tentativa, por exemplo `closureAttemptId`, armazenar o execution ARN e definir a estratégia de nomes. Usar apenas `tenantId` já não é uma idempotency key correta quando o lifecycle admite múltiplos ciclos `ACTIVE ↔ HELD_FOR_RECOVERY`.

4. A classificação de status em `CloseOrganizationService` ficaria errada.

O comentário e a implementação preservam uma invariante importante: se a transição DynamoDB comitou e `StartExecution` falhou, uma repetição deve relançar a execução ([close-organization.ts:17](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/modules/organization/application/close-organization.ts:17)).

2. `StartExecution` falha;
3. a repetição vê `HELD_FOR_RECOVERY`;
4. o guard exaustivo em [close-organization.ts:100](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/modules/organization/application/close-organization.ts:100) rejeita o fechamento;
5. o tenant fica permanentemente em hold sem execução que encerre a janela.

A proposta não identifica esse bug, embora afirme que a implementação seria apenas “1 estado + 1 Wait + 1 service”.

5. A mudança do grafo é maior e mais perigosa do que descrito.

Não basta acrescentar `HELD_FOR_RECOVERY` ao union e a uma tabela: é necessário alterar uma defesa global deliberadamente codificada.

Além disso, `HELD` já significa estado operacional/jurídico bloqueado, com `blockedFrom` e retomada ao mesmo estado ([tenant-lifecycle-record.ts:46](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/shared/tenant-lifecycle/tenant-lifecycle-record.ts:46)). - legal hold durante a janela;
- erro operacional durante cancelamento;
- `BLOCKED`;
- retomada;

6. `HELD_FOR_RECOVERY` bloqueia mutações de negócio — mas bloqueia também o próprio caminho normal de cancelamento.

A parte positiva é que o fence transacional continuará bloqueando corretamente mutações comuns: `TENANT_ACTIVE_STATUS` permanece exclusivamente `ACTIVE` em [tenant-lifecycle-record.ts:62](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/shared/tenant-lifecycle/tenant-lifecycle-record.ts:62). Portanto, enquanto o estado for `HELD_FOR_RECOVERY`, escritas que usam `TenantBusinessMutation` não devem cometer.

Mas a proposta não distingue:

- mutação de negócio, que deve continuar proibida;
- mutação de lifecycle do sistema;
- cancelamento administrativo excepcional.

Se o cancelamento for implementado por um primitive de business mutation, ele será corretamente recusado pelo fence. Se usar `SystemMutationStore`, precisa de contrato OCC explícito e autorização especial anterior à resolução tenant-ativa. Nada disso foi decidido.

7. A alteração RBAC está incompleta.

`Action` é union fechado em [authorization.ts:7](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/modules/identity/domain/authorization.ts:7), e `ACTION_ROLES` é `Record<Action, ...>` em [authorization.ts:128](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/modules/identity/domain/authorization.ts:128). A proposta menciona “1 ação RBAC”, mas não registra as mudanças obrigatórias no union, matriz, testes de allowlist e auditoria de negação.

Mais importante: adicionar a action à allowlist não resolve o bloqueio anterior de `RequestContext`. O problema não está apenas em `authorize()`.

8. A justificativa dos 30 dias é aceitável, mas contém uma afirmação objetivamente errada.

Trinta dias não é o “ponto médio” da faixa de 20–90 dias ([proposta:90](C:/Users/Usuario/Desktop/projects/expiration-tracker/docs/architecture/reviews/quarantine-retention-scoping/round-1-claude-proposal.md:90)); o ponto médio é 55. O prazo pode ser escolhido por ser o menor precedente convergente relevante e por alinhar-se aos prazos internos de retenção, mas a justificativa deve dizer isso corretamente.

Também há mistura conceitual: “encerramento + 30 dias” da matriz LGPD é prazo de retenção após o evento, não necessariamente uma janela de recuperação do encerramento. O número pode coincidir; a proposta não demonstrou que as duas políticas têm a mesma finalidade.

9. A generalização cross-cutting é só uma intenção.

A proposta chama o mecanismo de genérico, mas a única implementação descrita é `CancelOrganizationClosureService`, rota de Organization e `tenant-purge.asl.json`. Não define o contrato reutilizável mínimo:

- evento que inicia a quarentena;
- deadline persistido;
- identidade da tentativa;
- autoridade que pode recuperar;
- semântica de cancelamento;
- resolução de corrida no deadline;
- regra de legal hold;
- garantia de que o purger verifica novamente o estado antes da primeira exclusão física.

“Documentar o padrão” é razoável, mas o padrão ainda não foi especificado.

10. A prioridade LGPD é razoável, porém mistura classes com trabalhos heterogêneos.

`CORE_USER_DATA` em primeiro lugar é amplo demais para virar uma unidade implementável. A proposta deveria decompor por entidade e evento gerador de `purgeAfter`. `ACCOUNT_ACTIVE` também não é simplesmente “purga por idade dentro de tenant ativo”: a matriz diz “encerramento + 30 dias” ([privacy-lgpd.md:37](C:/Users/Usuario/Desktop/projects/expiration-tracker/docs/architecture/privacy-lgpd.md:37)), então cada entidade precisa de semântica concreta de encerramento.

Colocar `LEGAL_EVIDENCE` em quarto na “ordem de implementação” também é enganoso: essa classe está bloqueada por aprovação jurídica, KMS independente e Object Lock ([privacy-lgpd.md:40](C:/Users/Usuario/Desktop/projects/expiration-tracker/docs/architecture/privacy-lgpd.md:40)). Ela deveria aparecer numa lane bloqueada separada, não consumir uma posição linear entre workers executáveis.

11. O escopo “implementável na mesma sessão” está gravemente subestimado.

Não são apenas um estado, um Wait, um service, uma rota e uma action. O desenho correto exige, no mínimo:

- mudança do grafo e das invariantes;
- tentativa de fechamento identificável;
- persistência do deadline e execution ARN;
- estratégia de nome da execução;
- resolução de identidade especial para organização não ativa;
- cancelamento/compensação da Step Function;
- tratamento das corridas deadline versus cancel;
- atualização das classificações de status do close;
- mudanças de composição/IAM;
- testes de concorrência, idempotência e falhas parciais;
- atualização do sweeper e dos contratos ASL.

Logo, o critério 6 não foi satisfeito: há uma decisão explícita de escopo, mas baseada numa estimativa tecnicamente falsa.

A parte positiva é que o fence transacional continuará bloqueando corretamente mutações comuns: `TENANT_ACTIVE_STATUS` permanece exclusivamente `ACTIVE` em [tenant-lifecycle-record.ts:62](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/shared/tenant-lifecycle/tenant-lifecycle-record.ts:62). Portanto, enquanto o estado for `HELD_FOR_RECOVERY`, escritas que usam `TenantBusinessMutation` não devem cometer.

Mas a proposta não distingue:

- mutação de negócio, que deve continuar proibida;
- mutação de lifecycle do sistema;
- cancelamento administrativo excepcional.

Se o cancelamento for implementado por um primitive de business mutation, ele será corretamente recusado pelo fence. Se usar `SystemMutationStore`, precisa de contrato OCC explícito e autorização especial anterior à resolução tenant-ativa. Nada disso foi decidido.

7. A alteração RBAC está incompleta.

`Action` é union fechado em [authorization.ts:7](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/modules/identity/domain/authorization.ts:7), e `ACTION_ROLES` é `Record<Action, ...>` em [authorization.ts:128](C:/Users/Usuario/Desktop/projects/expiration-tracker/src/modules/identity/domain/authorization.ts:128). A proposta menciona “1 ação RBAC”, mas não registra as mudanças obrigatórias no union, matriz, testes de allowlist e auditoria de negação.

Mais importante: adicionar a action à allowlist não resolve o bloqueio anterior de `RequestContext`. O problema não está apenas em `authorize()`.

8. A justificativa dos 30 dias é aceitável, mas contém uma afirmação objetivamente errada.

Trinta dias não é o “ponto médio” da faixa de 20–90 dias ([proposta:90](C:/Users/Usuario/Desktop/projects/expiration-tracker/docs/architecture/reviews/quarantine-retention-scoping/round-1-claude-proposal.md:90)); o ponto médio é 55. O prazo pode ser escolhido por ser o menor precedente convergente relevante e por alinhar-se aos prazos internos de retenção, mas a justificativa deve dizer isso corretamente.

Também há mistura conceitual: “encerramento + 30 dias” da matriz LGPD é prazo de retenção após o evento, não necessariamente uma janela de recuperação do encerramento. O número pode coincidir; a proposta não demonstrou que as duas políticas têm a mesma finalidade.

9. A generalização cross-cutting é só uma intenção.

A proposta chama o mecanismo de genérico, mas a única implementação descrita é `CancelOrganizationClosureService`, rota de Organization e `tenant-purge.asl.json`. Não define o contrato reutilizável mínimo:

- evento que inicia a quarentena;
- deadline persistido;
- identidade da tentativa;
- autoridade que pode recuperar;
- semântica de cancelamento;
- resolução de corrida no deadline;
- regra de legal hold;
- garantia de que o purger verifica novamente o estado antes da primeira exclusão física.

“Documentar o padrão” é razoável, mas o padrão ainda não foi especificado.

10. A prioridade LGPD é razoável, porém mistura classes com trabalhos heterogêneos.

`CORE_USER_DATA` em primeiro lugar é amplo demais para virar uma unidade implementável. A proposta deveria decompor por entidade e evento gerador de `purgeAfter`. `ACCOUNT_ACTIVE` também não é simplesmente “purga por idade dentro de tenant ativo”: a matriz diz “encerramento + 30 dias” ([privacy-lgpd.md:37](C:/Users/Usuario/Desktop/projects/expiration-tracker/docs/architecture/privacy-lgpd.md:37)), então cada entidade precisa de semântica concreta de encerramento.

Colocar `LEGAL_EVIDENCE` em quarto na “ordem de implementação” também é enganoso: essa classe está bloqueada por aprovação jurídica, KMS independente e Object Lock ([privacy-lgpd.md:40](C:/Users/Usuario/Desktop/projects/expiration-tracker/docs/architecture/privacy-lgpd.md:40)). Ela deveria aparecer numa lane bloqueada separada, não consumir uma posição linear entre workers executáveis.

11. O escopo “implementável na mesma sessão” está gravemente subestimado.

Não são apenas um estado, um Wait, um service, uma rota e uma action. O desenho correto exige, no mínimo:

- mudança do grafo e das invariantes;
- tentativa de fechamento identificável;
- persistência do deadline e execution ARN;
- estratégia de nome da execução;
- resolução de identidade especial para organização não ativa;
- cancelamento/compensação da Step Function;
- tratamento das corridas deadline versus cancel;
- atualização das classificações de status do close;
- mudanças de composição/IAM;
- testes de concorrência, idempotência e falhas parciais;
- atualização do sweeper e dos contratos ASL.

Logo, o critério 6 não foi satisfeito: há uma decisão explícita de escopo, mas baseada numa estimativa tecnicamente falsa.
