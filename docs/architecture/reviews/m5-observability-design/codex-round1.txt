import { defaultRedactor, Redactor } from "./redactor.js";

export type LogLevel = "debug" | "info" | "warn" | "error";

export interface LogContext {
  correlationId?: string;
  tenantId?: string;
  requestId?: string;
  [key: string]: unknown;
}

export interface SecureLoggerOptions {
  redactor?: Redactor;
  /** Base context merged into every log line (e.g. service name, function name). */
  baseContext?: LogContext;
  /** Injectable sink for tests; defaults to console. */
  sink?: (level: LogLevel, line: string) => void;
  now?: () => string;
}

const defaultSink = (level: LogLevel, line: string): void => {
  // eslint-disable-next-line no-console -- this IS the sink; central logger is the sanctioned exception.
  const fn = level === "error" ? console.error : level === "warn" ? console.warn : console.log;
  fn(line);
};

export class SecureLogger {
  private readonly redactor: Redactor;
  private readonly baseContext: LogContext;
  private readonly sink: (level: LogLevel, line: string) => void;
  private readonly now: () => string;

  constructor(options: SecureLoggerOptions = {}) {
    this.redactor = options.redactor ?? defaultRedactor;
    this.baseContext = options.baseContext ?? {};
    this.sink = options.sink ?? defaultSink;
    this.now = options.now ?? (() => new Date().toISOString());
  }

  /** Returns a new logger with additional persistent context (e.g. bind correlationId once per invocation). */
  child(context: LogContext): SecureLogger {
    return new SecureLogger({
      redactor: this.redactor,
      baseContext: { ...this.baseContext, ...context },
      sink: this.sink,
      now: this.now,
    });
  }

  debug(event: string, context: LogContext = {}): void {
    this.write("debug", event, context);
  }

  info(event: string, context: LogContext = {}): void {
    this.write("info", event, context);
  }

  warn(event: string, context: LogContext = {}): void {
    this.write("warn", event, context);
  }

  error(event: string, context: LogContext = {}): void {
    this.write("error", event, context);
  }

  private write(level: LogLevel, event: string, context: LogContext): void {
    const merged = { ...this.baseContext, ...context };
    const redacted = this.redactor.redact(merged) as Record<string, unknown>;
    const line = JSON.stringify({
      timestamp: this.now(),
      level,
      event,
      ...redacted,
    });
    this.sink(level, line);
  }
}

export async function handler(event: SQSEvent): Promise<SQSBatchResponse> {
  const batchItemFailures: { itemIdentifier: string }[] = [];

      const parsed: unknown = JSON.parse(record.body);
      const { valid, errors } = defaultSchemaRegistry.validate(DISPATCH_COMMAND_SCHEMA_ID, parsed);
      if (!valid) {
        // Schema-invalid payload is treated as a poison message, not a retryable failure -
        // it can never become valid on retry. Still reported as a batch item failure so
        // SQS's own maxReceiveCount+DLQ (not this handler) is what stops the retries
        // (m3.5-runtime-design.md Â§10 "Poison message").
        logger.error("reminder-dispatch schema-invalid payload", { messageId: record.messageId, errors });
        batchItemFailures.push({ itemIdentifier: record.messageId });
        continue;

      const command = parsed as DispatchCommand;
      const outcome = await dispatchOccurrence(deps, command);
      logger.info("reminder-dispatch outcome", { messageId: record.messageId, outcome: outcome.kind });
    } catch (err) {
      logger.error("reminder-dispatch failed", { messageId: record.messageId, error: err instanceof Error ? err.message : String(err) });

        logger.error("email-delivery schema-invalid payload", { messageId: record.messageId, errors });
        batchItemFailures.push({ itemIdentifier: record.messageId });
        continue;
      }
      const envelope = parsed as EmailDeliverEnvelope;
      const command: EmailDeliverCommandData = {
        tenantId: envelope.tenantId,
        intentId: envelope.data.intentId,
        attemptId: envelope.data.attemptId,
        itemId: envelope.data.itemId,
        expectedItemVersion: envelope.data.expectedItemVersion,
        templateId: envelope.data.templateId,
        templateVersion: envelope.data.templateVersion,
        locale: envelope.data.locale,
        deliverNotBefore: envelope.data.deliverNotBefore,
        correlationId: envelope.correlationId,
      };
      const outcome = await processEmailDelivery(deps, command);
      logger.info("email-delivery outcome", { messageId: record.messageId, attemptId: command.attemptId, outcome: outcome.kind });
      if (outcome.kind === "DEFERRED") {
        // Quiet hours not over yet - never discard. The router already scheduled a
        // one-shot EventBridge Scheduler re-delivery (design Â§5.3); reporting THIS delivery
        // as a batch item failure would just requeue it needlessly before that fires. No
        // action needed here beyond the log above.
        continue;
      }
    } catch (err) {
      logger.error("email-delivery failed", { messageId: record.messageId, error: err instanceof Error ? err.message : String(err) });
      batchItemFailures.push({ itemIdentifier: record.messageId });
    }
  }

  return { batchItemFailures };
}

 * not in the workflow, so the workflow stays testable without SES/SNS-specific parsing. */
import type { SQSBatchResponse, SQSEvent } from "aws-lambda";
import { createDocumentClient } from "../../../shared/dynamodb/client.js";
import { buildSesCallbackDeps } from "../composition/notification.js";
import { processSesCallback, type ParsedSesCallbackEvent } from "../../../modules/notification/application/ses-callback-workflow.js";
import type { SesCallbackEventKind } from "../../../modules/notification/application/ses-callback-processor.js";
import { SecureLogger } from "../../../shared/observability/logger.js";

const tableName = process.env["TABLE_NAME"];
const providerAccountId = process.env["SES_ACCOUNT_ALIAS"];
if (!tableName) throw new Error("TABLE_NAME env var is required.");
if (!providerAccountId) throw new Error("SES_ACCOUNT_ALIAS env var is required.");
const deps = buildSesCallbackDeps(client, tableName, providerAccountId);
const logger = new SecureLogger({ baseContext: { service: "ses-callback" } });

interface SnsEnvelope {
  Type: string;
  MessageId: string;
  TopicArn: string;
  Message: string;
}

interface SesEventJson {
  eventType: "Delivery" | "Bounce" | "Complaint" | "Reject" | "Rendering Failure";
  mail: { messageId: string; timestamp: string; tags?: Record<string, string[]> };
}

const EVENT_KIND_BY_SES_TYPE: Partial<Record<SesEventJson["eventType"], SesCallbackEventKind>> = {
  Delivery: "DELIVERY",
  Bounce: "BOUNCE",
  Complaint: "COMPLAINT",
};

function firstTag(tags: Record<string, string[]> | undefined, name: string): string | undefined {
  return tags?.[name]?.[0];
}

/** Never throws on malformed input - an unparseable message is logged and skipped (not
 * retried into a redrive loop that will never succeed), never crashes the whole batch. */
function parseSesCallback(body: string): ParsedSesCallbackEvent | undefined {
  const envelope = JSON.parse(body) as SnsEnvelope;
  const sesEvent = JSON.parse(envelope.Message) as SesEventJson;
  const eventKind = EVENT_KIND_BY_SES_TYPE[sesEvent.eventType];
  if (!eventKind) return undefined; // Reject/Rendering Failure - not handled as attempt transitions in M4

  return {
    snsMessageId: envelope.MessageId,
    eventKind,
    providerMessageId: sesEvent.mail.messageId,
    tags: {
      attemptId: firstTag(sesEvent.mail.tags, "et_attempt_id"),
      intentId: firstTag(sesEvent.mail.tags, "et_intent_id"),
      tenantId: firstTag(sesEvent.mail.tags, "et_tenant_id"),
    },
    occurredAt: sesEvent.mail.timestamp,
  };
}

export async function handler(event: SQSEvent): Promise<SQSBatchResponse> {
  const batchItemFailures: { itemIdentifier: string }[] = [];

  for (const record of event.Records) {
    try {
      const parsedEvent = parseSesCallback(record.body);
      if (!parsedEvent) {
        logger.info("ses-callback unhandled event type, skipping", { messageId: record.messageId });
        continue;
      }
      const outcome = await processSesCallback(deps, parsedEvent);
      logger.info("ses-callback outcome", { messageId: record.messageId, outcome: outcome.kind });
    } catch (err) {
      logger.error("ses-callback failed", { messageId: record.messageId, error: err instanceof Error ? err.message : String(err) });
      batchItemFailures.push({ itemIdentifier: record.messageId });
    }
  }

  return { batchItemFailures };
}

NOTA: 6,8/10

1. A propagação X-Ray por SQS está tecnicamente errada. Com SQS instrumentado e Lambda com tracing ativo, o contexto é propagado e os traces do produtor/consumidor são ligados automaticamente por AWSTraceHeader, um system attribute reservado. Gravar apenas trace_id em MessageAttributes.traceId perde parent ID e decisão de sampling; substituir o facade segment gerenciado pela Lambda com setSegment(new Segment(...)) também viola o modelo do runtime e pode gerar traces órfãos ou corrompidos. Fonte oficial: [Amazon SQS and AWS X-Ray](https://docs.aws.amazon.com/xray/latest/devguide/xray-services-sqs.html).

2. A proposta está desatualizada em relação à infraestrutura real. infra/modules/lambda-function já possui tracing_active=true por default, tracing_config Active e AWSXRayDaemonWriteAccess condicional. Portanto, apresentar esses itens como implementação nova de M5 mascara que D-022 já está parcialmente implementada e fará a execução/testes mirarem uma mudança inexistente. O design deve registrar explicitamente o baseline e limitar o delta à instrumentação/verificação que realmente falta.

3. O escopo de AsyncLocalStorage está incorreto para handlers batch. “Uma chamada no topo do handler” cria um contexto por invocação, mas os handlers SQS e DynamoDB Streams processam vários records, potencialmente de tenants e correlations diferentes. Não existe um único record.messageAttributes.correlationId que possa inicializar esse contexto. Cada record precisa de runWithContext próprio, abrangendo parse, workflow e logs; isso também torna segura uma futura paralelização do batch.

4. A atualização tardia de tenantId não está definida por uma API segura. getContext() + merge não altera o store do AsyncLocalStorage, salvo se o objeto retornado for mutado diretamente; mutação implícita cria semântica frágil e permite vazamento de contexto entre itens se o escopo batch permanecer como proposto. O design precisa decidir entre contexto imutável aninhado por runWithContext, enterWith cuidadosamente delimitado, ou uma API explícita de bind/update, com testes de isolamento concorrente.

5. A propagação de correlationId pelo outbox está quebrada conceitualmente. Os SendMessage reais ocorrem no relay ou sweeper, em uma invocação posterior disparada por DynamoDB Streams/EventBridge. getContext() nesse momento contém a correlação do relay/sweeper, não a da operação que criou o OutboxRecord. A correlação original precisa ser persistida no envelope/outbox e o sender deve extraí-la desse payload; adicionar somente MessageAttributes no momento do envio não preserva causalidade ponta a ponta.

6. O desenho não cobre corretamente fontes assíncronas que não são SQS. NotificationRouter e os outbox relays recebem DynamoDB Streams, mas a tabela de resolução só define API Gateway, SQS e Scheduler. Sem regra por record para Streams — preferencialmente usando correlationId persistido no item/outbox e fallback para eventID — parte central do pipeline continuará recebendo correlações novas e desconectadas.

7. A escolha de aws-xray-sdk-core cria retrabalho previsível. Desde 25 de fevereiro de 2026 o SDK X-Ray entrou em maintenance mode e a própria AWS recomenda migração para OpenTelemetry. D-022 aprovou X-Ray como backend, não obrigatoriamente o SDK legado; além disso, a afirmação de que OTel exigiria operar Collector em ECS/Fargate é falsa para Lambda, pois ADOT pode exportar para X-Ray por layer. O design deve comparar AWS Distro for OpenTelemetry/Lambda com o SDK legado antes de adicionar uma dependência já em manutenção. Fonte oficial: [AWS X-Ray SDK for Node.js](https://docs.aws.amazon.com/xray/latest/devguide/xray-sdk-nodejs.html).

8. O destino SNS→email não fecha sozinho o requisito de “alerta real”. Subscription email permanece PendingConfirmation até confirmação manual; Terraform não consegue confirmar pelo destinatário. Falta definir esse passo operacional, owner, critério de aceite e um teste controlado de alarme até a caixa postal. Sem isso, o deploy pode ficar verde enquanto nenhum alerta é entregue.

9. O plano de testes está ausente para as propriedades mais arriscadas. São necessários, no mínimo: isolamento ALS entre records/tenants, ausência de contexto residual após exceção, preservação da correlação via outbox/relay/sweeper, partial batch failure, trace ligado SQS→Lambda em sandbox e confirmação/entrega do alerta SNS. Sem esses critérios, os erros dos achados 1, 3 e 5 provavelmente só aparecerão em operação.

10. Concordo com manter src/workers observability-agnostic: contexto, instrumentação e logging pertencem às bordas/composition roots e preservam E-007.

11. Concordo com SNS→email como primeiro canal: é proporcional ao estágio pré-produção e permite adicionar PagerDuty/Slack posteriormente sem redesenhar os alarmes.

12. Concordo com não usar tenantId como dimensão de métrica: evita cardinalidade/custo descontrolados e mantém o isolamento de dados operacionais.
