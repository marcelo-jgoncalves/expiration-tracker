export interface ScopedLambdaFunctionProps extends Omit<lambda.FunctionProps, "code" | "handler"> {
  access: AccessCapability[];
  /** Real bundling path (M3.5): TS/JS entrypoint, bundled via esbuild (see bundleEntry above)
   * into a CJS asset. Mutually exclusive with `code` (tests use `code` for inline synthetic
   * bodies; production handlers use `entry`). */
  entry?: string;
  /** Exported handler function name inside `entry`. Only "handler" is currently supported
   * (bundleEntry throws otherwise) - kept as an explicit prop rather than hardcoded so a
   * synth-time error names the mismatch instead of silently invoking the wrong export. */
  handlerExport?: string;
  /** Test-only escape hatch for inline/synthetic Lambda code. Mutually exclusive with `entry`. */
  code?: lambda.Code;
}

/**
 * Every Lambda in this system MUST be constructed via ScopedLambdaFunction, never
 * `new lambda.Function` directly (implementation-blueprint.md §17.1) - it is the single
 * enforcement point for least-privilege IAM per function.
 */
export class ScopedLambdaFunction extends Construct {
  readonly function: lambda.Function;

  constructor(scope: Construct, id: string, props: ScopedLambdaFunctionProps) {
    super(scope, id);

    for (const capability of props.access) {
      if (!KNOWN_CAPABILITY_KINDS.has(capability.kind)) {
        // "falha synth quando o recurso solicitado não corresponde a uma capability conhecida"
        throw new Error(
          `ScopedLambdaFunction(${id}): unknown access capability kind "${capability.kind}". ` +
            `Known kinds: ${[...KNOWN_CAPABILITY_KINDS].join(", ")}.`,
        );
      }
    }

    const { access, code, entry, handlerExport, ...rest } = props;

    if (entry && code) {
      throw new Error(`ScopedLambdaFunction(${id}): "entry" and "code" are mutually exclusive.`);
    }

    const resolvedCode = entry
      ? bundleEntry(entry, handlerExport ?? "handler")
      : (code ?? lambda.Code.fromInline("exports.handler = async () => ({ statusCode: 501 });"));

    this.function = new lambda.Function(this, "Function", {
      timeout: Duration.seconds(10),
      memorySize: 256,
      tracing: lambda.Tracing.ACTIVE,
      ...rest,
      handler: "index.handler",
      code: resolvedCode,
    });

    for (const capability of access) {
      capability.grant(this.function);
    }
  }
}

## Decisão central: outbox durável entre claim e publicação

`TransactWrite(SCHEDULED→CLAIMED)` seguido de `SendMessage` direto no handler do producer **não é atômico** — se a Lambda cai entre os dois passos, a ocorrência fica `CLAIMED` sem mensagem em fila, e isso NÃO é coberto pela janela de lookback do próprio producer (que consulta `M` outra chave a cada tick) nem necessariamente pela reconciliação diária a tempo. Fila+DLQ sozinhas não fecham essa lacuna (achado que nenhuma das duas propostas independentes tinha isoladamente — só apareceu na crítica cruzada).

**Decisão**: o claim grava, na MESMA `TransactWriteItems`, um `OutboxEvent` (reusando `shared/outbox/outbox.ts` — nenhum mecanismo de outbox paralelo), `eventType: "ReminderDispatchRequested"`, com um atributo novo e obrigatório **`destination: "SQS_REMINDER_DISPATCH_V1"`** — o único discriminador de roteamento. `OutboxPublisher` (M2/M3, genérico → EventBridge) só processa itens SEM `destination` ou com `destination` que ele reconhece como "EventBridge padrão"; ao encontrar `destination: "SQS_REMINDER_DISPATCH_V1"` ele **ignora explicitamente** o item (não é "o padrão dele, só que falhando" — é um `if` textual no seu filtro de query/processamento). Só o `DispatchOutboxRelay` Lambda processa itens com esse `destination`. Isso é testado diretamente: um item com `destination: "SQS_REMINDER_DISPATCH_V1"` nunca aparece publicado no EventBridge, e um item sem `destination` nunca é consumido pelo relay SQS.

`DispatchOutboxRelay`: gatilho DynamoDB Streams (`NEW_IMAGE`, filtro no event source mapping por `destination`), `reportBatchItemFailures` (partial batch failure do próprio Stream). Para cada registro: (1) lease condicional — `UpdateItem` com `ConditionExpression: attribute_not_exists(leaseOwner) OR leaseExpiresAt < :now`, evita dois relays concorrentes publicando o mesmo item em paralelo; (2) `SendMessage` para `ReminderDispatchQueue`; (3) `UpdateItem` transição `PENDING→PUBLISHED` só após `SendMessage` confirmado. Falha entre (2) e (3) é aceitável — duplicata é absorvida pela idempotência do dispatch (M3, claim por `occurrenceId` determinístico) — falha antes de (2) deixa o item `PENDING`, recuperável.

**Sweeper** (`OutboxSweeperReminderDispatch`, reusa o mecanismo de sweeper já previsto para `OutboxPublisher` genérico, mesmo padrão): EventBridge Scheduler a cada 5 minutos, varre `GSI6PK=RECON#OUTBOX#PENDING` filtrando `destination: "SQS_REMINDER_DISPATCH_V1"` e `occurredAt` mais antigo que um limiar (2 min — cobre falha do Stream/relay, não falha transitória normal), reenvia via o mesmo caminho do relay (lease condicional inclusive). Retry/backoff: nenhum backoff próprio — cada execução do sweeper é uma tentativa; itens que falham repetidamente continuam `PENDING` e acumulam idade, visível em alarme de "outbox pendente há mais de N execuções do sweeper" (reusa o alarme de outbox já documentado em `implementation-blueprint.md`).

## Reconciliação: precisa de fonte real de candidatos, entra no escopo

`runReconciliation` (M3) recebe candidatos como parâmetro injetado — rodar a Lambda via EventBridge sem uma query real produzindo esses candidatos seria false-green (exatamente o P1 #3 do red team). **Correção de convergência (Rodada 4 de validação)**: a primeira versão desta seção usava `GSI6PK=TENANT#t#WORKSTATE#<status>` (tenant-scoped), mas isso é **inconsultável pelo reconciliador global** — o mesmo erro estrutural já cometido e corrigido no GSI3 (`implementation-blueprint.md` §9.2): `Query` exige `PK` exato, e o job de reconciliação não sabe a priori quais tenants têm claim expirado ou candidato DST. `RECON#OUTBOX#PENDING` (outbox, seção anterior) já resolve isso com uma chave GLOBAL sem tenant — é o padrão a seguir aqui também, não uma exceção nova.

**Decisão**: `GSI6PK` **global** (sem `tenantId`), `tenantId` preservado no `GSI6SK` e no item base para reconstrução de contexto — chave global cria blast radius global deliberado nas funções que a consultam, mesmo trade-off do GSI3, mitigado pela mesma disciplina de IAM explícito (`infra/lib/dynamo-table.ts`, `PolicyStatement`s por recurso, nunca `grantReadData` genérico):

- **Exatamente duas roles** recebem `dynamodb:Query` sobre `<tableArn>/index/GSI6`: `ReminderReconciliation` e `OutboxSweeperReminderDispatch` (o sweeper também precisa consultar `RECON#OUTBOX#PENDING`, que já vive neste índice — não é exclusividade de uma única função, é uma lista fechada de duas, nenhuma tenant-facing).
- Nenhuma outra role (incluindo `ReminderProducer`, `ReminderDispatch`, `RemindersHandler`, `ItemsHandler`) recebe esse grant, mesmo que já tenham `gsi3Read()` ou acesso de tabela — capability nova e distinta em `scoped-lambda-function.ts` (`gsi6Read()`, mesmo padrão de `gsi3Read()`), nunca implícita em `readWriteKeys`/`readKeys`.
- Teste de synth (`test/infra/stack.test.ts`) verifica positivamente que só essas duas roles referenciam `/index/GSI6`, e negativamente que nenhuma outra o faz — mesmo padrão do teste de isolamento do GSI3 (M3).
- Teste de sandbox AWS (Camada 3) exige `AccessDenied` tanto para GSI3 quanto para GSI6 ao assumir a role de uma função tenant-facing.
- Remoção das chaves `GSI6PK`/`GSI6SK` (`WORKSTATE#CLAIMED`/`WORKSTATE#DST_PENDING`) ocorre **atomicamente na mesma `TransactWriteItems`/`UpdateItem` da mudança de estado correspondente** (claim→triggered, claim revertido, janela DST processada) — nunca como passo separado que pode falhar independente e deixar o índice inconsistente com o estado real.

Escrita das chaves (sem isso, ter só a fonte de leitura não torna o índice correto):
- `GSI6PK="WORKSTATE#CLAIMED"`/`GSI6SK` gravados pelo `ReminderProducer` na MESMA `TransactWriteItems` que faz `SCHEDULED→CLAIMED` (builder novo em `occ.ts`, SET, mesmo padrão do GSI3 shard); removidos (ou o builder ganha suporte a REMOVE, gap já documentado em M3 §judgment call 2 — reavaliar aqui, já que agora este ponteiro passa a ser lido ativamente, não só tolerado como órfão) quando o `ReminderDispatch` faz `CLAIMED→TRIGGERED`, e quando a `ReminderReconciliation` reverte `CLAIMED→SCHEDULED` por claim expirado.
- `GSI6PK="WORKSTATE#DST_PENDING"`/`GSI6SK` gravados pelo `ReminderMaterializer` (M3, já existente) ao criar/atualizar uma `ReminderOccurrence` cuja janela de recorrência cruza uma transição de DST conhecida (mesma lógica IANA-aware de M3, só passa a persistir o resultado como candidato indexado em vez de recalcular do zero); removidos quando a reconciliação DST processa a janela.

Namespace `WORKSTATE#CLAIMED`/`WORKSTATE#DST_PENDING` (global, sem tenant) não colide com `RECON#OUTBOX#PENDING` (global, outbox) nem com `TENANT#t#WORKSTATE#<status>` de outros domínios (Document/Upload, M5+, que continuam tenant-scoped por não precisarem de reconciliação cross-tenant) — são famílias de chave `GSI6PK` distintas por prefixo, coexistindo no mesmo índice físico. Port separado de `ReminderStore` (mesmo princípio de isolamento que levou a `ReminderProducerStore` a ser separado — não dar a rotas tenant-facing acesso a uma query que elas não precisam). Isso muda o escopo original: o "wiring real do GSI6" (item 4 da lista de pendências pós-M3) deixa de ser follow-up separado e entra neste milestone — sem ele, a EventBridge Rule/Scheduler de reconciliação (item 4 do escopo de G8) dispara sobre nada.

Cadência: reconciliação de **claims expirados** roda em minutos (curta — recupera leases/claims excepcionais), separada da reconciliação **DST/materialização diária** (mesmo handler, campo `mode` no evento, concurrency reservada 1 preservada); são responsabilidades complementares ao sweeper do outbox (que recupera publicação pendente), não redundantes.

/**
 * Central redactor - implementation-blueprint.md #14.1, threat-model.md redactor gap.
 * Loads the denylist from schemas/sensitive-fields.json (single source of truth shared
 * with any future documentation/tooling) and scrubs field names + known value patterns
 * before anything reaches logs, trace annotations, DLQ diagnostic metadata or operational
 * events.
 */
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import path from "node:path";

interface SensitiveFieldsConfig {
  redactedFieldNames: string[];
  redactedValuePatterns: { name: string; pattern: string }[];
  limits: {
    maxDepth: number;
    maxStringLength: number;
    maxArrayLength: number;
    maxObjectKeys: number;
  };
  redactedValuePlaceholder: string;
}

const PLACEHOLDER_FALLBACK = "[REDACTED]";
const TRUNCATED_SUFFIX = "…[TRUNCATED]";

function resolveSchemaPath(): string {
  // src/shared/observability -> repo root/schemas/sensitive-fields.json
  const here = path.dirname(fileURLToPath(import.meta.url));
  return path.resolve(here, "../../../schemas/sensitive-fields.json");
}

function loadConfig(explicitPath?: string): SensitiveFieldsConfig {
  const configPath = explicitPath ?? resolveSchemaPath();
  const raw = readFileSync(configPath, "utf-8");
  return JSON.parse(raw) as SensitiveFieldsConfig;
}

export interface RedactorOptions {
  /** Override for tests; defaults to schemas/sensitive-fields.json relative to this file. */
  configPath?: string;
}

export class Redactor {
  private readonly fieldNames: Set<string>;
  private readonly valuePatterns: { name: string; regex: RegExp }[];
  private readonly limits: SensitiveFieldsConfig["limits"];
  private readonly placeholder: string;

  constructor(options: RedactorOptions = {}) {
    const config = loadConfig(options.configPath);
    this.fieldNames = new Set(config.redactedFieldNames.map((f) => f.toLowerCase()));
    this.valuePatterns = config.redactedValuePatterns.map((p) => ({
      name: p.name,
      regex: new RegExp(p.pattern, "gi"),
    }));
    this.limits = config.limits;
    this.placeholder = config.redactedValuePlaceholder ?? PLACEHOLDER_FALLBACK;
  }

  /** Redacts an arbitrary structured value (object/array/primitive) recursively. */
  redact(value: unknown): unknown {
    return this.redactValue(value, 0, new WeakSet());
  }

  /** Redacts known value patterns (emails, bearer tokens, ...) inside a free-text string,
   * e.g. sanitized SDK/provider exception messages (implementation-blueprint.md #14.1). */
  redactString(value: string): string {
    let result = this.truncateString(value);
    for (const { regex } of this.valuePatterns) {
      regex.lastIndex = 0;
      result = result.replace(regex, this.placeholder);
    }
    return result;
  }

  /** Redacts an Error, producing a plain object safe to log (never logs raw `.stack` of
   * SDK errors, which frequently embed request bodies/headers). */
  redactError(err: unknown): Record<string, unknown> {
    if (err instanceof Error) {
      return {
        name: err.name,
        message: this.redactString(err.message),
        // Stack traces can leak values interpolated into error messages (SDKs do this a lot);
        // keep only the first line (type + message) to preserve some debuggability.
        stackSummary: this.redactString(err.stack?.split("\n")[0] ?? err.message),
      };
    }
    return { message: this.redactString(String(err)) };
  }

  private truncateString(value: string): string {
    if (value.length <= this.limits.maxStringLength) {
      return value;
    }
    return value.slice(0, this.limits.maxStringLength) + TRUNCATED_SUFFIX;
  }

  private redactValue(value: unknown, depth: number, seen: WeakSet<object>): unknown {
    if (depth > this.limits.maxDepth) {
      return "[MAX_DEPTH_EXCEEDED]";
    }
    if (value === null || value === undefined) {
      return value;
    }
    if (typeof value === "string") {
      return this.redactString(value);
    }
    if (typeof value === "number" || typeof value === "boolean") {
      return value;
    }
    if (value instanceof Error) {
      return this.redactError(value);
    }
    if (Array.isArray(value)) {
      if (seen.has(value)) return "[CIRCULAR]";
      seen.add(value);
      return value.slice(0, this.limits.maxArrayLength).map((v) => this.redactValue(v, depth + 1, seen));
    }
    if (typeof value === "object") {
      if (seen.has(value as object)) return "[CIRCULAR]";
      seen.add(value as object);
      const entries = Object.entries(value as Record<string, unknown>).slice(0, this.limits.maxObjectKeys);
      const out: Record<string, unknown> = {};
      for (const [key, v] of entries) {
        if (this.fieldNames.has(key.toLowerCase())) {
          out[key] = this.placeholder;
        } else {
          out[key] = this.redactValue(v, depth + 1, seen);
        }
      }
      return out;
    }
    // functions, symbols, bigint etc - never log these verbatim
    return "[UNSUPPORTED_TYPE]";
  }
}

## Proposta independente — Eixo 4: Segurança da Informação e AppSec

O eixo avalia se o sistema preserva confidencialidade, integridade, disponibilidade e isolamento diante de abuso intencional, comprometimento parcial e falhas nas fronteiras de confiança. Não reavalia qualidade geral do código, arquitetura, cobertura de testes ou disciplina documental, já pertencentes aos outros eixos; considera esses elementos apenas quando constituem controles ou evidências especificamente de segurança.

Pesos calibrados pelo risco real do projeto: multi-tenancy em single-table, GSIs globais GSI3/GSI6 com blast radius deliberado, identidades Cognito/JWT e pipeline assíncrono pelo menos uma vez. A referência normativa combina [OWASP ASVS](https://owasp.org/www-project-application-security-verification-standard/), [OWASP Top 10](https://owasp.org/Top10/), [OWASP SAMM](https://owaspsamm.org/model/) e as práticas de [least privilege do AWS IAM](https://docs.aws.amazon.com/IAM/latest/UserGuide/access_policies.html) e [DynamoDB](https://docs.aws.amazon.com/amazondynamodb/latest/developerguide/rbac-best-practices.html).
